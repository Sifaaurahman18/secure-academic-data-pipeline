# Kebutuhan pustaka Secure Academic Data Pipeline

# Dipasang di dalam virtual environment:
#
#   python -m venv .venv
#   source .venv/Scripts/activate      # Git Bash di Windows
#   .venv\Scripts\activate             # Command Prompt di Windows
#   python -m pip install --upgrade pip
#   python -m pip install -r requirements.txt
#
# Catatan untuk dosen: setelah starter diuji pada komputer laboratorium,
# kunci versinya dengan `python -m pip freeze > requirements.lock.txt`
# agar seluruh kelas memakai versi yang sama sepanjang semester.
